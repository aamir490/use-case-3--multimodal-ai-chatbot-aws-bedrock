# ── AWS SDK ──────────────────────────────────────────────────
# boto3 is the AWS SDK — used for Bedrock, STS, and S3 calls.
# botocore is boto3's low-level foundation (installed automatically).

botocore>=1.34.0


# ── LangChain (retained from original project) ───────────────
# The original chatbot_backend.py uses LangChain. Kept so that
# chatbot_frontend.py and chatbot_frontend2.py continue to work.

langchain-aws>=1.0.0
langchain-core>=0.3.0
langchain-community>=0.3.0


# ── Streamlit ─────────────────────────────────────────────────

# and st.file_uploader type filtering.
